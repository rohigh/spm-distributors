rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Check if the logged-in user is an admin (case-sensitive, must be lowercase)
    function isAdmin() {
      return request.auth != null && 
             (request.auth.token.email == "info@spmdistributors.co.uk" || 
              request.auth.token.email == "17sixesrohigh@gmail.com");
    }

    // Helper function to check if user has an APPROVED wholesale account
    function isApprovedWholesaleUser() {
      let userDoc = get(/databases/$(database)/documents/wholesale_users/$(request.auth.uid));
      return request.auth != null && userDoc != null && userDoc.data.status == 'APPROVED';
    }

    // Helper function to get the user's wholesale category
    function getUserCategory() {
      return get(/databases/$(database)/documents/wholesale_users/$(request.auth.uid)).data.category;
    }

    // =========================================
    // WHOLESALE USERS COLLECTION
    // =========================================
    match /wholesale_users/{userId} {
      // A user can read/write their own pending registration data
      allow read, create: if request.auth != null && request.auth.uid == userId;
      // Only admins can update the status or category, or delete users
      allow update, delete: if isAdmin();
      // Admins can read all wholesale users
      allow read: if isAdmin();
    }

    // =========================================
    // WHOLESALE PRICING COLLECTIONS
    // =========================================
    match /prices_dover/{productId} {
      allow read: if isApprovedWholesaleUser() && getUserCategory() == 'DOVER';
      allow read, write: if isAdmin();
    }

    match /prices_kent/{productId} {
      allow read: if isApprovedWholesaleUser() && getUserCategory() == 'KENT';
      allow read, write: if isAdmin();
    }

    match /prices_long_distance/{productId} {
      allow read: if isApprovedWholesaleUser() && getUserCategory() == 'LONG_DISTANCE';
      allow read, write: if isAdmin();
    }

    // =========================================
    // WHOLESALE CATALOG (Products & Categories)
    // =========================================
    match /wholesale_categories/{categoryId} {
      allow read: if isApprovedWholesaleUser() || isAdmin();
      allow write: if isAdmin();
    }
    
    match /wholesale_products/{productId} {
      allow read: if isApprovedWholesaleUser() || isAdmin();
      allow write: if isAdmin();
    }

    // =========================================
    // RETAIL CATALOG (Products & Categories)
    // =========================================
    match /categories/{categoryId} {
      allow read: if true;
      allow write: if isAdmin();
    }
    
    match /products/{productId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // =========================================
    // ORDERS COLLECTION
    // =========================================
    match /orders/{orderId} {
      // Anyone can create an order
      allow create: if true;
      
      // Allow customers to view their own receipt using the secret order ID
      allow get: if true; 
      
      // ONLY admins can view the list of all orders, update, or delete them
      allow list, update, delete: if isAdmin();
    }
    
    // =========================================
    // SETTINGS COLLECTION
    // =========================================
    match /settings/{docId} {
      // Anyone can read settings
      allow read: if true;
      
      // ONLY admins can change settings
      allow write: if isAdmin();
    }
  }
}
